cmake_minimum_required(VERSION 3.16)
project(m24e2d_sdk_worker_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(M24E2D_ENABLE_HARDWARE_SDK "Build physical Z1 SDK code; DOES NOT QUALIFY HARDWARE" OFF)
set(B2B "${CMAKE_CURRENT_LIST_DIR}/../m24e2b_z1_lowcmd_adapter")
if(NOT EXISTS "${B2B}/include/z1_lowcmd_plan.h" OR NOT EXISTS "${B2B}/src/z1_lowcmd_plan.cpp")
  message(FATAL_ERROR "M2.4e2b planner not installed beside M2.4e2d")
endif()
add_executable(m24e2d_z1_worker src/z1_worker_dual.cpp "${B2B}/src/z1_lowcmd_plan.cpp")
target_include_directories(m24e2d_z1_worker PRIVATE "${B2B}/include")
target_compile_options(m24e2d_z1_worker PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(M24E2D_ENABLE_HARDWARE_SDK)
  set(SDK_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../thirdparty/z1_sdk" CACHE PATH "Vendor Z1 SDK root")
  if(NOT EXISTS "${SDK_ROOT}/include/unitree_arm_sdk/control/unitreeArm.h")
    message(FATAL_ERROR "Official SDK header missing under: ${SDK_ROOT}")
  endif()
  set(SDK_LIB "${SDK_ROOT}/lib/libZ1_SDK_${CMAKE_SYSTEM_PROCESSOR}.so")
  if(NOT EXISTS "${SDK_LIB}")
    message(FATAL_ERROR "Official SDK shared library missing: ${SDK_LIB}")
  endif()
  find_package(Threads REQUIRED)
  target_sources(m24e2d_z1_worker PRIVATE src/z1_sdk_bridge.cpp)
  target_include_directories(m24e2d_z1_worker BEFORE PRIVATE "${CMAKE_CURRENT_LIST_DIR}/include")
  target_include_directories(m24e2d_z1_worker SYSTEM PRIVATE "${SDK_ROOT}/include")
  target_compile_definitions(m24e2d_z1_worker PRIVATE M24E2D_ENABLE_HARDWARE_SDK M24E2B_ENABLE_HARDWARE_SDK)
  target_link_libraries(m24e2d_z1_worker PRIVATE "${SDK_LIB}" Threads::Threads)
  message(WARNING "Compiled physical Z1 SDK path; hardware NOT validated. Do NOT run on real arm until bench procedure approved.")
else()
  message(STATUS "SAFE DEFAULT: physical Z1 SDK backend not compiled or linked")
endif()
