cmake_minimum_required(VERSION 3.16)
project(m24e2e_live_input LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}")
find_package(Threads REQUIRED)
set(REPO_ROOT "" CACHE PATH "Path to user's unitree_rl_mjlab (empty uses fixture ONLY)")
add_library(m24e2e_ipc STATIC src/z1_ipc_client.cpp)
target_include_directories(m24e2e_ipc PUBLIC include)
target_link_libraries(m24e2e_ipc PUBLIC Threads::Threads)
target_compile_options(m24e2e_ipc PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(m24e2e_client_test tests/test_client.cpp)
target_link_libraries(m24e2e_client_test PRIVATE m24e2e_ipc)
add_executable(m24e2e_mock_j5_control_test tests/test_mock_j5_control.cpp)
target_link_libraries(m24e2e_mock_j5_control_test PRIVATE m24e2e_ipc)
if(REPO_ROOT)
  set(INT "${REPO_ROOT}/deploy/interactive_b2z1")
  foreach(f interactive_observation.h interactive_feedback_adapter.h)
    if(NOT EXISTS "${INT}/include/${f}")
      message(FATAL_ERROR "Required original Interactive header missing: ${f}")
    endif()
  endforeach()
  add_library(m24e2e_original STATIC
    "${INT}/src/interactive_feedback_adapter.cpp"
    "${INT}/src/interactive_observation.cpp")
  target_include_directories(m24e2e_original PUBLIC "${INT}/include")
else()
  message(WARNING "Fixture mode: tests ONLY compile a minimal API replica; use -DREPO_ROOT=/your/repo for real-source parity")
  add_library(m24e2e_original STATIC tests/fixture/adapter_fixture.cpp tests/fixture/interactive_observation.cpp)
  target_include_directories(m24e2e_original PUBLIC tests/fixture)
endif()
add_library(m24e2e_live STATIC src/b2z1_live_input.cpp)
target_include_directories(m24e2e_live PUBLIC include)
target_link_libraries(m24e2e_live PUBLIC m24e2e_original m24e2e_ipc)
target_compile_options(m24e2e_live PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(m24e2e_live_test tests/test_live_input.cpp)
target_link_libraries(m24e2e_live_test PRIVATE m24e2e_live)
